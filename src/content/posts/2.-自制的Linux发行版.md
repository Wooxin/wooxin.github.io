## 准备工作
```shell
# 创建一个目录用于存放工作文件
mkdir -p mymini-linux/rootfs mymini-linux/kernel
cd /mymini-linux
```

## 编译内核
```shell
# 下载并编译最新的Linux内核
# 自己去找最新内核
wget https://www.kernel.org/pub/linux/kernel/v7.x/linux-7.0.4.tar.gz
tar -xf linux-7.0.4.tar.gz
cd linux-7.0.4

# 编译
# 生成最小化配置
make defconfig
# 或
# 自定义配置
make menuconfig
# 将GPU WIFI Bluetooth全都关闭
# 关闭Enable loadable module support

# 如果你是Archlinux, 需要关闭内核签名
scripts/config --disable SYSTEM_TRUSTED_KEYS
scripts/config --disable SYSTEM_REVOCATION_KEYS
scripts/config --disable CONFIG_DEBUG_INFO_BTF
# 编译
make -j$(nproc)

# 将模块安装到rootfs
make modules_install INSTALL_MOD_PATH=../rootfs
# 将内核镜像复制到kernel目录
cp ./arch/x86/boot/bzImage ../kernel

# 删除无用软件与刚才编译的包节省空间
cd ../
rm -fr ./linux*
```

## 编译BusyBox
使用BusyBox可以一站式使用Linux基础命令, 从而取代GNU工具链

```shell
# 下载解压, 自行寻找最新版
wget https://busybox.net/downloads/busybox-1.37.0.tar.bz2
tar -xf busybox-1.37.0.tar.bz2

# 编译
cd busybox-1.36.1
make defconfig  # 默认配置
make menuconfig  # 自定义：启用 init、sh 等，禁用多余（如 httpd, td）
# 如果你是archlinux一定会遇到ncurses报错, 通过修改./scripts/kconfig/lxdialog/check-lxdialog.sh中的main() {} 为int main() {}来解决
make -j$(nproc)
make install CONFIG_PREFIX=../rootfs  # 安装到 rootfs
cd ..
rm -fr busybox-1.37.0*
```

## 构建根文件系统 (rootfs)
### 创建必要的目录和文件
```shell
cd rootfs
mkdir -p bin sbin etc/init.d dev proc sys tmp var lib
# 创建设备节点
sudo mknod dev/console c 5 1
sudo mknod dev/null c 1 3
sudo mknod dev/zero c 1 5
# 创建 init 脚本（/etc/init.d/rcS 或直接用 BusyBox init）
echo '#!/bin/sh' > etc/init.d/rcS
echo 'mount -t proc none /proc' >> etc/init.d/rcS
echo 'mount -t sysfs none /sys' >> etc/init.d/rcS
echo 'mount -t tmpfs none /tmp' >> etc/init.d/rcS
echo '/sbin/mdev -s' >> etc/init.d/rcS  # 动态设备
chmod +x etc/init.d/rcS
ln -s bin/busybox init  # 用 BusyBox 作为 init
cd ..
```

## 打包成镜像并测试
### 创建磁盘镜像
```shell
dd if=/dev/zero of=rootfs.img bs=1M count=20  # 20M镜像
mkfs.ext4 rootfs.img
mkdir ./rootfstestmk
mount rootfs.img ./rootfstestmk
cp -a rootfs/* ./rootfstestmk
umount ./rootfstestmk

# 或 (只读)
grub-mkrescue -o woxminimal-V1.09.iso ./rootfs/
```
